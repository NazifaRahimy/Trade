// REGISTER
// ======================================================

export const registerUser = async (
  email: string,
  password: string
) => {
  const generatedUsername =
    email.split("@")[0];

  const payload = {
    username: generatedUsername,
    email: email,
    password: password,
  };

  const response = await api.post(
    "/api/auth/register/",
    payload
  );

  return response.data;
};
